from datetime import UTC, datetime
from langchain_core.runnables import RunnableConfig

def _now_iso() -> str:

async def clarification_gate(state: AgentState, config: RunnableConfig) -> dict:

    Deterministic, no LLM. Produces a clarification_request and final_response candidate.

            "provider_latency_ms": None, "retry_count": 0, "metrics_json": None}
    return {
        "clarification_request": request,
        "final_response": "Could you provide a bit more information so I can help?",

    }
