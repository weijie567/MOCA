```python
from __future__ import annotations

from typing import Any

from src.agent.state import AgentState

MIN_EVIDENCE_SCORE = 0.55  # mirror retrieve_policy_evidence.MIN_EVIDENCE_SCORE
_FACT_ONLY_INTENTS = {"order_status_inquiry"}
_PERMISSION_CODES = {"FORBIDDEN", "permission_denied"}


def _intent(state: AgentState) -> str:
    return state.get("primary_intent") or state.get("current_intent") or "unknown"


def _denial_blocks_required_claims(
    denied_resources: set[str | None],
    claim_dependency_map: Any,
    missing: list[str],
    facts: Any,
    retrieval_status: Any,
    intent: str,
) -> bool:
    """Fail closed unless typed dependencies preserve an independent answer path."""
    if not isinstance(claim_dependency_map, list) or not claim_dependency_map:
        return True
    if any(
        not isinstance(entry, dict)
        or not isinstance(entry.get("claim_id"), str)
        or not isinstance(entry.get("depends_on_refs"), list)
        or any(
            not isinstance(ref, dict)
            or not isinstance(ref.get("resource_type"), str)
            or not isinstance(ref.get("resource_id"), str)
            for ref in entry.get("depends_on_refs", [])
        )
        for entry in claim_dependency_map
    ):
        return True
    denied_refs = [
        ref
        for entry in claim_dependency_map
        for ref in entry["depends_on_refs"]
        if ref.get("resource_id") in denied_resources or ref.get("resource_type") in denied_resources
    ]
    if not denied_refs:
        return False

    missing_required = {str(item) for item in missing if isinstance(item, str)}
    denied_ref_tokens = {
        token
        for ref in denied_refs
        for token in (ref.get("resource_type"), ref.get("resource_id"))
        if isinstance(token, str)
    }
    if missing_required & denied_ref_tokens:
        return True

    has_independent_facts = bool(facts)
    has_independent_policy_evidence = retrieval_status in {"strong_evidence", "partial_evidence"}
    if intent in _FACT_ONLY_INTENTS:
        return not has_independent_facts
    return not (has_independent_facts or has_independent_policy_evidence)


def route_after_investigate(state: AgentState) -> str:
    """Deterministic, side-effect-free router after the investigate node (§9.5).

    Reads STATE only (D-11). Returns a key in the route_after_investigate mapping.
    """
    _bc = state.get("business_context")
    bc = _bc if isinstance(_bc, dict) else {}  # type guard: non-dict truthy values (e.g. "not-a-dict") must not reach bc.get()
    missing = bc.get("missing_required_facts") or []
    errors = bc.get("errors") or []
    facts = bc.get("facts") or bc  # tolerate live shape where ctx IS the facts dict
    retrieval_status = state.get("retrieval_status")
    best_score = state.get("best_score")
    claim_dependency_map = state.get("claim_dependency_map") or []
    intent = _intent(state)

    # 1. permission denied -> resolve claim dependencies and fail closed where unverifiable
    denied = [
        e
        for e in errors
        if isinstance(e, dict) and (e.get("error_code") or e.get("code")) in _PERMISSION_CODES
    ]
    if denied:
        denied_resources = {e.get("resource") or e.get("resource_id") for e in denied}
        # Use claim_dependency_map to identify denied-dependent claims. Missing/invalid/
        # unverifiable mappings treat affected claims as denied. Short-circuit only when
        # those claims are required/answer-blocking; preserve independent valid facts.
        if _denial_blocks_required_claims(
            denied_resources, claim_dependency_map, missing, facts, retrieval_status, intent
        ):
            return "final_response"

    # 2. required facts still missing after investigation -> clarify
    if missing:
        return "clarification_gate"

    # 3. fact-only intent with facts obtained -> business_fact_response
    if intent in _FACT_ONLY_INTENTS and facts:
        return "final_response"

    # 4. retrieval insufficient/failed -> insufficient_evidence_response
    if retrieval_status in {"error", "no_evidence", None} or (
        isinstance(best_score, (int, float)) and best_score < MIN_EVIDENCE_SCORE
    ):
        return "final_response"

    # 5. sufficient investigation context
    return "recommendation_generation"
```
