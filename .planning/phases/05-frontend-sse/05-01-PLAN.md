创建 src/api/routers/agent_runs.py，实现以下 4 个端点：

**端点 1: POST /（创建 run）**
- 接受 CreateRunRequest (query, thread_id)
- 生成 run_id = str(uuid.uuid4())
- 调用 write_agent_run(session, run_id=run_id, thread_id=body.thread_id, tenant_id=str(user.tenant_id), user_id=str(user.id), input_query=body.query, final_status="pending", final_response=None, started_at=datetime.now(timezone.utc), completed_at=None, total_latency_ms=None)
- await session.commit()
- 返回 ApiResponse(success=True, data={"run_id": run_id, "status": "pending"}, trace_id=...)

**端点 2: GET /{run_id}（run 状态）**
- 解析 run_id UUID（用 _parse_run_id）
- 用 TraceRepository.get_run(run_uuid, user.tenant_id) 查询
- 404 if not found，403 if ownership check fails（同 traces.py 模式）
- 返回 ApiResponse(success=True, data=RunStatusResponse(run_id=..., final_status=..., started_at=..., completed_at=..., final_response=...).model_dump(mode="json"), trace_id=...)

**端点 3: GET /{run_id}/events（SSE 流）**
- 解析 run_id，查询 run，做 ownership check（同上）
- 构造 input_state 和 config（完全复用 agent.py 模式，但 query 和 thread_id 从 run 记录读取：run.input_query 和 run.thread_id）
- 注意：AgentRun 没有 input_query 字段暴露给 TraceRepository，需要直接查询 AgentRun 模型获取 input_query
- 返回 EventSourceResponse(event_generator())

event_generator() 逻辑：
```python
async def event_generator():
    import json
    from datetime import datetime, timezone
    now = lambda: datetime.now(timezone.utc).isoformat()
    
    # 1. 发出 run_started
    yield {"data": json.dumps({"event_type": "run_started", "run_id": run_id_str, "step_index": 0, "status": "running", "message": "正在接收请求", "timestamp": now(), "payload": {}})}
    
    step_index = 0
    trace_steps: list[dict] = []
    final_state: dict = {}
    
    try:
        async for node_name, update in graph.astream(input_state, config, stream_mode="updates"):
            step_index += 1
            # step_started
            message = NODE_MESSAGES.get(node_name, f"正在执行 {node_name}")
            yield {"data": json.dumps({"event_type": "step_started", "run_id": run_id_str, "step_index": step_index, "node_name": node_name, "status": "running", "message": message, "timestamp": now(), "payload": {}})}
            
            # 提取 payload
            payload = _extract_step_payload(node_name, update)
            
            # step_completed
            yield {"data": json.dumps({"event_type": "step_completed", "run_id": run_id_str, "step_index": step_index, "node_name": node_name, "status": "completed", "message": message, "timestamp": now(), "payload": payload})}
            
            # 累积 trace_steps 和 final_state
            if isinstance(update, dict):
                final_state.update(update)
                if "trace_steps" in update:
                    trace_steps = update["trace_steps"]
        
        # 正常完成
        final_response = final_state.get("final_response") or "处理完成。"
        yield {"data": json.dumps({"event_type": "final_response", "run_id": run_id_str, "step_index": step_index + 1, "status": "completed", "message": "已完成", "timestamp": now(), "payload": {"final_response": final_response}})}
        
        # 持久化
        total_ms = round((time.perf_counter() - t0) * 1000)
        completed_at = datetime.now(timezone.utc)
        final_status = build_trace_summary(run_id_str, final_state, total_ms).get("final_status", "completed")
        try:
            await write_agent_run(session, run_id=run_id_str, thread_id=run.thread_id, tenant_id=str(user.tenant_id), user_id=str(user.id), input_query=run.input_query, final_status=final_status, final_response=final_response, started_at=run.started_at, completed_at=completed_at, total_latency_ms=total_ms)
            if trace_steps:
                await write_agent_steps(session, run_id=run_id_str, trace_steps=trace_steps)
            await session.commit()
        except Exception:
            await session.rollback()
    
    except Exception as exc:
        if _is_graph_interrupt(exc):
            # 处理 approval_required
            interrupt_data = _extract_interrupt_data(exc)
            approval_repo = ApprovalRepository(session)
            expires_at = _parse_expires_at(interrupt_data.get("expires_at"))
            approval = await approval_repo.create(
                run_id=UUID(run_id_str), tenant_id=user.tenant_id, requested_by=user.id,
                proposed_action=interrupt_data.get("proposed_action") or {},
                risk_level=interrupt_data.get("risk_level") or "high",
                risk_rule_ref=interrupt_data.get("risk_rule_ref"),
                risk_reason=interrupt_data.get("risk_reason"),
                expires_at=expires_at, thread_id=run.thread_id,
            )
            await approval_repo.add_step(approval.id, event_type="created", actor_id=user.id)
            # 更新 run 状态为 interrupted
            try:
                await write_agent_run(session, run_id=run_id_str, thread_id=run.thread_id, tenant_id=str(user.tenant_id), user_id=str(user.id), input_query=run.input_query, final_status="interrupted", final_response=None, started_at=run.started_at, completed_at=datetime.now(timezone.utc), total_latency_ms=round((time.perf_counter() - t0) * 1000))
                await session.commit()
            except Exception:
                await session.rollback()
            yield {"data": json.dumps({"event_type": "approval_required", "run_id": run_id_str, "step_index": step_index + 1, "status": "waiting_approval", "message": "需要审批，等待人工决策", "timestamp": now(), "payload": {"approval_id": str(approval.id), "proposed_action": interrupt_data.get("proposed_action"), "risk_level": interrupt_data.get("risk_level")}})}
        else:
            yield {"data": json.dumps({"event_type": "error", "run_id": run_id_str, "step_index": step_index + 1, "status": "failed", "message": "执行遇到问题，请重试", "timestamp": now(), "payload": {"error_message": str(exc)}})}
            try:
                await write_agent_run(session, run_id=run_id_str, thread_id=run.thread_id, tenant_id=str(user.tenant_id), user_id=str(user.id), input_query=run.input_query, final_status="error", final_response=None, started_at=run.started_at, completed_at=datetime.now(timezone.utc), total_latency_ms=round((time.perf_counter() - t0) * 1000))
                await session.commit()
            except Exception:
                await session.rollback()
```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client→SSE endpoint | Bearer token 必须在 HTTP header 中传递，不允许 URL query token |
| SSE generator→LangGraph | graph.astream() 输出的 node_name 和 update 来自内部，但 payload 提取需防御性处理 |
| run ownership | 跨 tenant 或跨 user 的 run 访问必须被拒绝 |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-05-01 | Spoofing | GET /agent-runs/{run_id}/events | mitigate | Security(get_current_user, scopes=["agent:chat"]) 验证 JWT；tenant_id 从 token 提取，不接受客户端传入 |
| T-05-02 | Information Disclosure | GET /agent-runs/{run_id} | mitigate | ownership check：run.user_id != user.id and user.role not in SUPERVISOR_ROLES → 403；cross-tenant 查询在 get_run(run_uuid, user.tenant_id) 层面被过滤 |
