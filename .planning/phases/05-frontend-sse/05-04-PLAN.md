  <action>
在 docker-compose.yml 的 api service 之后、volumes 之前添加 frontend service：

```yaml
  frontend:
    build:
      context: ./frontend
      dockerfile: Dockerfile
      target: dev
    ports:
      - "3000:3000"
    depends_on:
      api:
        condition: service_healthy
    environment:
      VITE_API_URL: http://api:8000
    volumes:
      - ./frontend/src:/app/src
      - ./frontend/index.html:/app/index.html
    healthcheck:
      test: ["CMD", "wget", "--spider", "-q", "http://localhost:3000"]
      interval: 10s
      timeout: 5s
      retries: 3
      start_period: 15s
```

volumes mount 确保开发时热更新生效。target: dev 使用 Dockerfile 的 dev stage。
