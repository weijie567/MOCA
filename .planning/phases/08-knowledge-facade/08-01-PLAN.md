</objective>

  from __future__ import annotations

  import hashlib
  import unicodedata

  EVIDENCE_TEXT_HASH_VERSION = "evidence_text_hash.v1"


  def normalize_evidence_text(text: str) -> str:
      """evidence_text_hash.v1 normalization: NFC, strip outer whitespace,
      unify internal newlines to \\n, NO case-folding."""
      normalized = unicodedata.normalize("NFC", text)
      normalized = normalized.replace("\r\n", "\n").replace("\r", "\n")
      normalized = normalized.strip()
      return normalized


  def evidence_text_hash(text: str) -> str:
      """Return sha256:<lowercase hex> of the normalized UTF-8 bytes."""
      digest = hashlib.sha256(normalize_evidence_text(text).encode("utf-8")).hexdigest()
      return f"sha256:{digest}"
  ```

  Do NOT case-fold. Do NOT collapse internal multiple spaces (the sub-spec only specifies newline unification + outer strip). Internal newline unification handles `\r\n` and `\r` before the strip.

  def canonical_evidence_projection(refs: list[EvidenceRefV1]) -> list[dict]:
      """Producer-side hash projection (NOT the Phase 13 hash profile):
      strip score, retain rank, rank-aware deterministic sort."""
      items = []
      for r in refs:
          d = r.model_dump()
          d.pop("score", None)  # score never enters hash material
          items.append(d)
      all_ranked = all(d.get("rank") is not None for d in items) and len(items) > 0
      if all_ranked:
          items.sort(key=lambda d: (d["rank"], d["evidence_id"], d["text_hash"]))
      else:
          items.sort(key=lambda d: (d["evidence_id"], d["text_hash"]))
      return items
  ```
