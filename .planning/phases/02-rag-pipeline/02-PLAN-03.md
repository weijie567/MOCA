class PolicyChunkRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def delete_by_document_id(self, doc_id: UUID, tenant_id: UUID) -> int:
        """Delete all chunks for a document. Returns count deleted."""
        stmt = delete(PolicyChunk).where(
            PolicyChunk.doc_id == doc_id,
            PolicyChunk.tenant_id == tenant_id,
        )
        result = await self.session.execute(stmt)
        return result.rowcount

    async def bulk_insert(self, chunks: list[PolicyChunk]) -> None:
        """Insert chunks in batch."""
        self.session.add_all(chunks)
        await self.session.flush()

    async def search_similar(
        self,
        query_embedding: list[float],
        tenant_id: UUID,
        top_k: int = 5,

    ) -> list[tuple[PolicyChunk, float]]:

        All within a single transaction — fail entire doc if any step fails.
        """
