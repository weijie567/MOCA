```python
from src.rag.schemas import EvidenceItem, RetrievalResult
from src.rag.embedder import EmbeddingService
from src.repositories.policy_chunk import PolicyChunkRepository

STRONG_EVIDENCE_THRESHOLD = 0.70
MIN_SIMILARITY_THRESHOLD = 0.55
FALLBACK_MESSAGE = "当前知识库中没有找到足够证据支持这个问题，建议转人工或补充规则文档。"

class Retriever:
    def __init__(self, chunk_repo: PolicyChunkRepository, embedder: EmbeddingService):
        self.chunk_repo = chunk_repo
        self.embedder = embedder

    async def search(
        self,
        query: str,
        tenant_id: UUID,
        top_k: int = 5,
        doc_type: str | None = None,
        risk_level: str | None = None,
    ) -> RetrievalResult:
        # 1. Embed query
        query_embedding = await self.embedder.embed_query(query)

        # 2. Search pgvector with metadata filters
        results = await self.chunk_repo.search_similar(
            query_embedding=query_embedding,
            tenant_id=tenant_id,
            top_k=top_k,
            min_similarity=MIN_SIMILARITY_THRESHOLD,
            doc_type=doc_type,
            risk_level=risk_level,
        )

        # 3. Build evidence items
        evidence = [
            EvidenceItem(
                doc_id=chunk.doc_id,  # semantic stable ID
                chunk_id=chunk.chunk_id,
                title=chunk.document.title,
                section=chunk.section,
                score=score,
                text=chunk.content[:200],  # excerpt
            )
            for chunk, score in results
        ]

        # 4. Determine retrieval_status
        best_score = max((e.score for e in evidence), default=0.0)
        if not evidence or best_score < MIN_SIMILARITY_THRESHOLD:
            status = "no_evidence"
        elif best_score >= STRONG_EVIDENCE_THRESHOLD:
            status = "strong_evidence"
        else:
            status = "partial_evidence"

        # 5. Return structured result
        return RetrievalResult(
            query=query,
            retrieval_status=status,
            evidence=evidence,
            best_score=best_score,
            fallback_message=FALLBACK_MESSAGE if status == "no_evidence" else None,
        )
```

Key behaviors:
- Thresholds: >=0.70 strong, 0.55-0.70 partial, <0.55 no_evidence (D-05)
- Returns structured RetrievalResult (D-09a)
- Each evidence item has doc_id, chunk_id, title, section, score, text excerpt (D-09b)
- Returns retrieval_status field (D-09c)
- Fallback message when no_evidence (D-05h)
- Retriever only returns evidence + score + status, does NOT generate answers (D-05i)

from src.rag.schemas import CitationValidation, RetrievalResult

def validate_citations(
    cited_chunk_ids: list[str],
    retrieval_result: RetrievalResult,
) -> CitationValidation:
    """
    Validate that all cited chunk_ids exist in the retrieval results.

    Rules (from D-06):
    - D-06a: Every policy answer must include citations
    - D-06b: Citation format: doc_id, chunk_id, title, section, score
    - D-06c: Answer with no citations -> invalid
    - D-06d: cited chunk_id not in retrieval results -> invalid
    - D-06e: Simple field matching (no LLM judge)
    """
    if not cited_chunk_ids:
        return CitationValidation(
            is_valid=False,
            invalid_citations=[],
            reason="No citations provided — every policy answer must include citations",
        )

    retrieved_ids = {e.chunk_id for e in retrieval_result.evidence}
    invalid = [cid for cid in cited_chunk_ids if cid not in retrieved_ids]

    if invalid:
        return CitationValidation(
            is_valid=False,
            invalid_citations=invalid,
            reason=f"Citations reference chunk_ids not in retrieval results: {invalid}",
        )

    return CitationValidation(is_valid=True)
```

from fastapi import APIRouter, Depends
from src.api.deps import get_current_user, get_db
from src.rag.schemas import SearchRequest, SearchResponse, RetrievalResult
from src.rag.retriever import Retriever
from src.rag.embedder import EmbeddingService

router = APIRouter(prefix="/search", tags=["search"])

@router.post("/", response_model=SearchResponse)
async def search_knowledge_base(
    request: SearchRequest,
    current_user = Depends(get_current_user),
    session = Depends(get_db),
):
    """
    Search knowledge base for relevant policy chunks.
    Scoped to current user's tenant_id.
    Returns structured evidence with confidence scoring.
    """
    embedder = EmbeddingService()
    chunk_repo = PolicyChunkRepository(session)
    retriever = Retriever(chunk_repo=chunk_repo, embedder=embedder)

    result = await retriever.search(
        query=request.query,
        tenant_id=current_user.tenant_id,
        top_k=request.top_k,
        doc_type=request.doc_type,
        risk_level=request.risk_level,
    )

    return SearchResponse(success=True, data=result)
```

Register router in src/api/main.py (add `from src.api.routers.search import router as search_router` and `app.include_router(search_router)`).

Key behaviors:
- Requires authentication (get_current_user dependency)
- Scopes search to current_user.tenant_id (D-10a, D-10b)
- Accepts optional doc_type and risk_level filters (RAG-04)
- Returns unified response format: {"success": true, "data": RetrievalResult}
