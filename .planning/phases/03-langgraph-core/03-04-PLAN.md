```python
"""Trace persistence: write AgentRun and AgentStep rows after graph.ainvoke().

Called by the API endpoint after each graph invocation. Per D-05.
"""
from __future__ import annotations

import uuid
from datetime import datetime, timezone
from typing import Any

from sqlalchemy.ext.asyncio import AsyncSession

from src.db.models import AgentRun, AgentStep


async def write_agent_run(
    session: AsyncSession,
    *,
    run_id: str,
    thread_id: str,
    tenant_id: str,
    user_id: str,
    input_query: str,
    final_status: str,
    final_response: str | None,
    started_at: datetime,
    completed_at: datetime,
    total_latency_ms: int,
    total_tokens: int | None = None,
    error_summary: str | None = None,
) -> AgentRun:
    """Insert one AgentRun row. Returns the persisted instance."""
    run = AgentRun(
        id=uuid.UUID(run_id),
        thread_id=thread_id,
        tenant_id=uuid.UUID(tenant_id),
        user_id=uuid.UUID(user_id),
        input_query=input_query,
        final_status=final_status,
        final_response=final_response,
        started_at=started_at,
        completed_at=completed_at,
        total_latency_ms=total_latency_ms,
        total_tokens=total_tokens,
        error_summary=error_summary,
    )
    session.add(run)
    await session.flush()  # get PK without committing
    return run


async def write_agent_steps(
    session: AsyncSession,
    *,
    run_id: str,
    trace_steps: list[dict[str, Any]],
) -> list[AgentStep]:
    """Insert one AgentStep row per entry in trace_steps. Returns persisted instances."""
    steps = []
    for idx, step in enumerate(trace_steps):
        started_at = _parse_dt(step.get("started_at"))
        completed_at = _parse_dt(step.get("completed_at"))
        agent_step = AgentStep(
            id=uuid.uuid4(),
            run_id=uuid.UUID(run_id),
            node_name=step["node"],
            step_index=idx,
            status=step.get("status", "completed"),
            input_summary=step.get("input_summary"),
            output_summary=step.get("output_summary"),
            tool_name=step.get("tool_name"),
            tool_input_summary=step.get("tool_input_summary"),
            tool_output_summary=step.get("tool_output_summary"),
            model_name=step.get("model_name"),
            prompt_tokens=step.get("prompt_tokens"),
            completion_tokens=step.get("completion_tokens"),
            latency_ms=step.get("latency_ms"),
            evidence_refs=step.get("evidence_refs"),
            error_message=step.get("error_message"),
            started_at=started_at or datetime.now(timezone.utc),
            completed_at=completed_at,
        )
        session.add(agent_step)
        steps.append(agent_step)
    await session.flush()
    return steps


def build_trace_summary(
    run_id: str,
    final_state: dict[str, Any],
    total_latency_ms: int,
) -> dict[str, Any]:
    """Build the trace_summary dict returned in the API response. Per D-05d.

    Fields: run_id, intent, nodes_executed, tools_called, evidence_count,
            risk_level, total_latency_ms, final_status.
    Does NOT expose: full prompts, full tool outputs, internal error stacks, PII.
    """
    trace_steps = final_state.get("trace_steps") or []
    nodes_executed = [s["node"] for s in trace_steps]
    tools_called = [s["tool_name"] for s in trace_steps if s.get("tool_name")]

    retrieved = final_state.get("retrieved_evidence") or {}
    evidence_count = len(retrieved.get("evidence") or [])

    risk = (final_state.get("risk_assessment") or {}).get("risk_level", "unknown")

    return {
        "run_id": run_id,
        "intent": final_state.get("current_intent", "unknown"),
        "nodes_executed": nodes_executed,
        "tools_called": tools_called,
        "evidence_count": evidence_count,
        "risk_level": risk,
        "total_latency_ms": total_latency_ms,
        "final_status": _derive_final_status(final_state),
    }


def _derive_final_status(state: dict[str, Any]) -> str:
    draft = state.get("recommendation_draft") or {}
    action = draft.get("recommended_action", "")
    if action in ("insufficient_evidence", "citation_invalid"):
        return "insufficient_evidence"
    errors = state.get("node_errors") or []
    if errors:
        return "error"
    if state.get("final_response"):
        return "completed"
    return "error"


def _parse_dt(value: str | None) -> datetime | None:
    if not value:
        return None
    try:
        return datetime.fromisoformat(value)
    except (ValueError, TypeError):
        return None
```

Create src/api/routers/agent.py:

```python
"""POST /api/v1/agent/chat — refund agent endpoint.

Auth: requires agent:chat scope.
Thread memory: caller supplies thread_id; same thread_id = same conversation.
Trace: writes AgentRun + AgentStep rows after each invocation.
"""
from __future__ import annotations

import time
import uuid
from datetime import datetime, timezone

from fastapi import APIRouter, Depends, Request, Security
from sqlalchemy.ext.asyncio import AsyncSession

from src.api.schemas.agent import ChatRequest, ChatResponse, TraceSummary
from src.api.schemas.common import ApiResponse, INTERNAL_ERROR
from src.auth.permissions import get_current_user
from src.db.models import User
from src.db.session import get_session
from src.agent.trace import write_agent_run, write_agent_steps, build_trace_summary

router = APIRouter(tags=["agent"])


@router.post("/chat", response_model=ApiResponse)
async def chat(
    body: ChatRequest,
    request: Request,
    session: AsyncSession = Depends(get_session),
    user: User = Security(get_current_user, scopes=["agent:chat"]),
) -> ApiResponse:
    """Submit a refund/order question to the agent. Returns evidence-cited response + trace."""
    graph = request.app.state.agent_graph
    started_at = datetime.now(timezone.utc)
    t0 = time.perf_counter()

    # Build initial state — persistent fields from JWT-validated user
    input_state = {
        "user_query": body.query,
        "thread_id": body.thread_id,
        "tenant_id": str(user.tenant_id),
        "user_id": str(user.id),
        "role": user.role,
    }

    # Config: thread_id for checkpointer + session for tool nodes
    config = {
        "configurable": {
            "thread_id": body.thread_id,
            "session": session,
        }
    }

    try:
        final_state = await graph.ainvoke(input_state, config)
    except Exception as exc:
        # Graceful degradation per D-10c, INFR-09
        total_ms = round((time.perf_counter() - t0) * 1000)
        fallback_response = "系统处理出现问题，请稍后重试或联系人工客服。"
        return ApiResponse(
            success=False,
            data={
                "response": fallback_response,
                "trace_summary": {
                    "run_id": str(uuid.uuid4()),
                    "intent": "unknown",
                    "nodes_executed": [],
                    "tools_called": [],
                    "evidence_count": 0,
                    "risk_level": "unknown",
                    "total_latency_ms": total_ms,
                    "final_status": "error",
                },
            },
            trace_id=request.state.trace_id,
        )

    total_ms = round((time.perf_counter() - t0) * 1000)
    completed_at = datetime.now(timezone.utc)

    run_id = final_state.get("current_run_id") or str(uuid.uuid4())
    trace_steps = final_state.get("trace_steps") or []
    final_response_text = final_state.get("final_response") or "处理完成，但未生成回复。"

    # Derive final_status for AgentRun row
    draft = final_state.get("recommendation_draft") or {}
    action = draft.get("recommended_action", "")
    if action in ("insufficient_evidence", "citation_invalid"):
        final_status = "insufficient_evidence"
    elif final_state.get("node_errors"):
        final_status = "error"
    else:
        final_status = "completed"

    # Count tokens from trace_steps
    total_tokens = sum(
        (s.get("prompt_tokens") or 0) + (s.get("completion_tokens") or 0)
        for s in trace_steps
    )

    # Persist trace — write then commit
    try:
        await write_agent_run(
            session,
            run_id=run_id,
            thread_id=body.thread_id,
            tenant_id=str(user.tenant_id),
            user_id=str(user.id),
            input_query=body.query,
            final_status=final_status,
            final_response=final_response_text,
            started_at=started_at,
            completed_at=completed_at,
            total_latency_ms=total_ms,
            total_tokens=total_tokens or None,
        )
        await write_agent_steps(session, run_id=run_id, trace_steps=trace_steps)
        await session.commit()
    except Exception:
        # Trace write failure must not fail the user response
        await session.rollback()

    trace_summary = build_trace_summary(run_id, final_state, total_ms)

    return ApiResponse(
        success=True,
        data=ChatResponse(
            response=final_response_text,
            trace_summary=TraceSummary(**trace_summary),
        ).model_dump(),
        trace_id=request.state.trace_id,
    )
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| HTTP request → agent endpoint | Untrusted query text; user identity from JWT only |
| graph output → trace writer | Final state may contain partial data; trace writer must handle None fields |
| trace writer → DB | Trace write failure must not fail the user response |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-03-10 | Information Disclosure | build_trace_summary | mitigate | trace_summary exposes only run_id, intent, node names, tool names, evidence_count, risk_level — no full prompts, no tool outputs, no PII per D-05e |
| T-03-11 | Spoofing | ChatRequest.thread_id | mitigate | thread_id is a conversation key, not an auth token; tenant isolation enforced by tenant_id from JWT in input_state, not from thread_id |
| T-03-12 | Denial of Service | graph.ainvoke timeout | mitigate | LLM timeout=30s per node; RetryPolicy max_attempts=2; outer try/except returns structured fallback response per INFR-09 |
| T-03-13 | Information Disclosure | trace write failure path | mitigate | session.rollback() on trace write failure; error not propagated to user response; internal error not exposed in API response |
</threat_model>
