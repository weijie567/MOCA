```python
from __future__ import annotations

from datetime import datetime
from typing import Any
from uuid import UUID

from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from src.db.models import ActionDraft, AgentRun, AgentStep, ApprovalRequest, ApprovalStep


class TraceRepository:
    def __init__(self, session: AsyncSession):
        self.session = session

    async def get_run(self, run_id: UUID, tenant_id: UUID) -> AgentRun | None:
        stmt = select(AgentRun).where(AgentRun.id == run_id, AgentRun.tenant_id == tenant_id)
        return (await self.session.execute(stmt)).scalar_one_or_none()

    async def get_steps(self, run_id: UUID) -> list[AgentStep]:
        stmt = select(AgentStep).where(AgentStep.run_id == run_id).order_by(AgentStep.step_index)
        return list((await self.session.execute(stmt)).scalars().all())

    async def get_approvals(self, run_id: UUID) -> list[ApprovalRequest]:
        stmt = select(ApprovalRequest).where(ApprovalRequest.run_id == run_id)
        return list((await self.session.execute(stmt)).scalars().all())

    async def get_approval_steps(self, approval_ids: list[UUID]) -> list[ApprovalStep]:
        if not approval_ids:
            return []
        stmt = select(ApprovalStep).where(ApprovalStep.approval_request_id.in_(approval_ids)).order_by(ApprovalStep.created_at)
        return list((await self.session.execute(stmt)).scalars().all())

    async def get_action_drafts(self, run_id: UUID) -> list[ActionDraft]:
        stmt = select(ActionDraft).where(ActionDraft.run_id == run_id)
        return list((await self.session.execute(stmt)).scalars().all())

    def build_timeline(
        self,
        steps: list[AgentStep],
        approvals: list[ApprovalRequest],
        approval_steps: list[ApprovalStep],
        drafts: list[ActionDraft],
    ) -> list[dict[str, Any]]:
        timeline: list[dict[str, Any]] = []

        for step in steps:
            timeline.append({
                "type": "agent_step",
                "time": step.started_at.isoformat(),
                "title": f"Node: {step.node_name}",
                "status": step.status,
                "detail": {
                    "node_name": step.node_name,
                    "tool_name": step.tool_name,
                    "latency_ms": step.latency_ms,
                    "provider_latency_ms": getattr(step, "provider_latency_ms", None),
                },
            })

        for approval in approvals:
            timeline.append({
                "type": "approval_request",
                "time": approval.created_at.isoformat(),
                "title": f"Approval requested: {approval.risk_rule_ref or 'unknown rule'}",
                "status": approval.status,
                "detail": {
                    "approval_id": str(approval.id),
                    "risk_level": approval.risk_level,
                    "proposed_action": approval.proposed_action,
                },
            })

        for astep in approval_steps:
            timeline.append({
                "type": "approval_decision",
                "time": astep.created_at.isoformat(),
                "title": f"Approval {astep.event_type}",
                "status": astep.event_type,
                "detail": {
                    "actor_id": str(astep.actor_id) if astep.actor_id else None,
                    "metadata": astep.metadata_json,
                },
            })

        for draft in drafts:
            timeline.append({
                "type": "action_draft",
                "time": draft.created_at.isoformat(),
                "title": f"Action: {draft.action_type}",
                "status": draft.status,
                "detail": {
                    "draft_id": str(draft.id),
                    "idempotency_key": draft.idempotency_key,
                },
            })

        timeline.sort(key=lambda x: x["time"])
        return timeline
```

```python
from __future__ import annotations

from uuid import UUID

from fastapi import APIRouter, Depends, HTTPException, Security
from sqlalchemy.ext.asyncio import AsyncSession

from src.api.schemas.approvals import TraceResponse
from src.api.schemas.common import ApiResponse
from src.auth.permissions import get_current_user
from src.db.models import User
from src.db.session import get_session
from src.repositories.trace_repo import TraceRepository

router = APIRouter(tags=["traces"])

SUPERVISOR_ROLES = {"supervisor", "admin", "approval_manager"}


@router.get("/{run_id}/trace", response_model=ApiResponse)
async def get_run_trace(
    run_id: str,
    session: AsyncSession = Depends(get_session),
    user: User = Security(get_current_user, scopes=["agent:chat"]),
) -> ApiResponse:
    repo = TraceRepository(session)
    run = await repo.get_run(UUID(run_id), user.tenant_id)

    if not run:
        raise HTTPException(status_code=404, detail={"code": "NOT_FOUND", "message": "Run not found"})

    # Permission: user sees own runs, supervisor/admin sees all in tenant
    if run.user_id != user.id and user.role not in SUPERVISOR_ROLES:
        raise HTTPException(status_code=403, detail={"code": "FORBIDDEN", "message": "Cannot view this run"})

    steps = await repo.get_steps(UUID(run_id))
    approvals = await repo.get_approvals(UUID(run_id))
    approval_ids = [a.id for a in approvals]
    approval_steps = await repo.get_approval_steps(approval_ids)
    drafts = await repo.get_action_drafts(UUID(run_id))
    timeline = repo.build_timeline(steps, approvals, approval_steps, drafts)

    trace_data = TraceResponse(
        run_id=str(run.id),
        thread_id=run.thread_id,
        input_query=run.input_query,
        final_status=run.final_status,
        final_response=run.final_response,
        started_at=run.started_at,
        completed_at=run.completed_at,
        total_latency_ms=run.total_latency_ms,
        steps=[{"node": s.node_name, "status": s.status, "latency_ms": s.latency_ms, "tool_name": s.tool_name} for s in steps],
        approvals=[...],  # Map to ApprovalResponse
        action_drafts=[{"id": str(d.id), "action_type": d.action_type, "status": d.status} for d in drafts],
        timeline=timeline,
    )

    return ApiResponse(success=True, data=trace_data.model_dump())
```
